    - deleteOldVersions(articleDbId: number, keepCount: number): DELETE FROM article_histories WHERE article_id = articleDbId AND version NOT IN (SELECT version FROM article_histories WHERE article_id = articleDbId ORDER BY version DESC LIMIT keepCount). Matches Go maxVersions=10
